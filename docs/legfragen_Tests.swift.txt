    /// Fachliche Korrekturen von Mario (6.9.2026) – dürfen nicht zurückfallen.

        let repo = ContentRepository()

            + repo.allChapters.flatMap { $0.questions }.map { $0.q + $0.options.joined() + $0.explanation }.joined()

    /// Vorlesestimme Conrad (zuerst nur „Legasthenie ist keine Krankheit oder Behinderung“): MP3 + Wortzeiten,
    /// Segment-Klartext muss exakt dem Swift-Parser entsprechen, sonst sitzt die Markierung daneben.
    func testVorlesenAlleKapitel() throws {
        let repo = ContentRepository()
        XCTAssertEqual(repo.allChapters.filter { $0.audio == nil }.count, 0, "jedes Kapitel hat Vorlesen")
        for c in repo.allChapters {
            let stem = try XCTUnwrap(c.audio, c.id)
            XCTAssertEqual(stem, c.bookSlug + "__" + c.file)
            XCTAssertNotNil(repo.audioURL(stem), "MP3 fehlt: \(stem)")
            let d = try XCTUnwrap(repo.vorleseDaten(stem), "Wortzeiten fehlen: \(stem)")
            XCTAssertEqual(d.voice, "de-DE-ConradNeural")
            XCTAssertGreaterThan(d.words.count, 300, c.id)
            XCTAssertGreaterThan(d.duration, 90_000, c.id)
            // Vorlesen beginnt ganz oben: Überschrift, Vorspann, dann der Text
            XCTAssertEqual(d.segments.first?.b, -2, c.id)
            XCTAssertEqual(d.segments.first?.text, c.title, c.id)
            XCTAssertEqual(d.segments.dropFirst().first?.b, -1, c.id)
            XCTAssertEqual(d.segments.dropFirst().first?.text, c.description, c.id)
            let blocks = MarkdownParser.parse(c.body)
            for seg in d.segments where seg.b >= 0 {
                var plain = ""
                switch blocks[seg.b] {
                case .heading(_, let t), .paragraph(let t), .quote(let t): plain = String(inlineMarkdown(t).characters)
                case .bullets(let items), .numbered(let items): plain = String(inlineMarkdown(items[seg.i]).characters)
                default: XCTFail("Segment zeigt auf Block ohne Text: \(seg.b) in \(c.id)")
                }
                XCTAssertEqual(plain, seg.text, "Klartext Block \(seg.b) in \(c.id)")
            }
            var vorher = -1
            for w in d.words {
                XCTAssertGreaterThanOrEqual(w.t, vorher, c.id); vorher = w.t
                XCTAssertTrue(w.a < w.e && w.e <= d.segments[w.s].text.count, "Wortbereich außerhalb: \(w) in \(c.id)")
            }
        }
        let c = try XCTUnwrap(repo.allChapters.first { $0.file == "02-legasthenie-dyskalkulie-keine-krankheit" })
        let d = try XCTUnwrap(repo.vorleseDaten(try XCTUnwrap(c.audio)))
        // Satzende-Markierungen: die Hervorhebung bleibt vor Punkt, Ausrufe- und Fragezeichen stehen
        let satzenden = d.words.filter { $0.p == true }
        XCTAssertGreaterThan(satzenden.count, 20)
        for w in satzenden {
            let text = d.segments[w.s].text
            let rest = text.dropFirst(w.e).drop { "“\"»«".contains($0) }
            XCTAssertTrue(rest.first.map { ".!?".contains($0) } ?? false, "kein Satzende nach: \(text.prefix(w.e).suffix(20))")
        }
    }

    func testAllImagesResolveAndNoOrphans() {
        let repo = ContentRepository()
        var referenced = Set<String>()
        for c in repo.allChapters {
            for b in MarkdownParser.parse(c.body) {
                if case .image(_, let asset) = b {
                    XCTAssertNotNil(repo.image(asset), "Bild fehlt: \(asset)")
                    referenced.insert(asset)
                }
            }
        }
        for e in repo.extras {
            for b in MarkdownParser.parse(e.body) {
                if case .image(_, let asset) = b {
                    XCTAssertNotNil(repo.image(asset), "Bild fehlt: \(asset)")
                    referenced.insert(asset)
                }
            }
        }
        XCTAssertFalse(referenced.isEmpty)
        let bundled = Set((Bundle.main.urls(forResourcesWithExtension: nil, subdirectory: "Content/Images") ?? [])
            .map(\.lastPathComponent))
        XCTAssertEqual(bundled.subtracting(referenced), [], "Bilder im Bundle ohne Kapitel (App schlank halten)")
    }

    func testMarkdownParser() {
        let md = """
        # Titel
        Erster Absatz
        über zwei Zeilen.

        ## Unter
        - eins
        - zwei
        1. a
        2. b
        ![Alt](x__y.jpg)
        > Zitat
        | A | B |
        |---|---|
        | 1 | 2 |
        ---
        """
        let b = MarkdownParser.parse(md)
        XCTAssertEqual(b.count, 9)
        if case .heading(let l, let t) = b[0] { XCTAssertEqual(l, 1); XCTAssertEqual(t, "Titel") } else { XCTFail() }
        if case .paragraph(let t) = b[1] { XCTAssertEqual(t, "Erster Absatz über zwei Zeilen.") } else { XCTFail() }
        if case .bullets(let i) = b[3] { XCTAssertEqual(i, ["eins", "zwei"]) } else { XCTFail() }
        if case .numbered(let i) = b[4] { XCTAssertEqual(i, ["a", "b"]) } else { XCTFail() }
        if case .image(let alt, let asset) = b[5] { XCTAssertEqual(alt, "Alt"); XCTAssertEqual(asset, "x__y.jpg") } else { XCTFail() }
        if case .quote(let t) = b[6] { XCTAssertEqual(t, "Zitat") } else { XCTFail() }
        if case .table(let rows) = b[7] { XCTAssertEqual(rows, [["A", "B"], ["1", "2"]]) } else { XCTFail() }
        if case .rule = b[8] {} else { XCTFail() }
    }

    func testProgressStore() {
        let d = UserDefaults(suiteName: "test.\(UUID().uuidString)")!
        let s = ProgressStore(defaults: d)
        XCTAssertFalse(s.isRead("a/b"))
        s.record("a/b", correct: 4, total: 5)
        XCTAssertTrue(s.isRead("a/b"))
        XCTAssertTrue(s.result("a/b")!.passed)
        s.record("a/b", correct: 2, total: 5)
        XCTAssertEqual(s.result("a/b")!.best, 4)
        XCTAssertEqual(s.result("a/b")!.last, 2)
        XCTAssertEqual(s.result("a/b")!.attempts, 2)
        XCTAssertEqual(s.accuracy!, 0.8, accuracy: 0.001)
        // Persistenz
        let s2 = ProgressStore(defaults: d)
        XCTAssertEqual(s2.result("a/b")!.best, 4)
        s2.reset()
        XCTAssertNil(ProgressStore(defaults: d).result("a/b"))
    }

    func testContinueChapter() {
        let repo = ContentRepository()
        let d = UserDefaults(suiteName: "test.\(UUID().uuidString)")!
        let s = ProgressStore(defaults: d)
        let first = repo.allChapters[0]
        XCTAssertEqual(s.continueChapter(repo)?.id, first.id)
        s.markRead(first.id)
        s.touch(first.id)
        XCTAssertEqual(s.continueChapter(repo)?.id, repo.allChapters[1].id)
        // Letztes Kapitel eines Themas gelesen → nächstes ungelesenes der Übersicht
        let last = repo.themes[0].chapters.last!
        s.markRead(last.id); s.touch(last.id)
        XCTAssertEqual(s.continueChapter(repo)?.id, repo.allChapters[1].id)
    }

    /// Bug Build 1: ein Pfad für alle Tabs – Quiz/Nächstes Kapitel aus dem Fortschritt-Tab liefen ins Leere.
    func testNavigatorUsesPathOfVisibleTab() {
        let nav = Navigator()
        nav.tab = 1
        nav.open(.thema("schulkinder"))
        nav.open(.chapter("a/b"))
        XCTAssertEqual(nav.progressPath.count, 2)
        XCTAssertEqual(nav.libraryPath.count, 0)
        nav.open(.quiz("a/b"))
        XCTAssertEqual(nav.progressPath.count, 3)
        nav.replace(last: 2, with: .chapter("a/c"))
        XCTAssertEqual(nav.progressPath.count, 2)
        XCTAssertEqual(nav.libraryPath.count, 0)

        nav.tab = 0
        nav.open(.thema("grundlagen"))
        XCTAssertEqual(nav.libraryPath.count, 1)
        XCTAssertEqual(nav.progressPath.count, 2)
        nav.back(5)
        XCTAssertEqual(nav.libraryPath.count, 0)
        XCTAssertEqual(nav.progressPath.count, 2)
        nav.tab = 2
        nav.open(.extra("30-jahre-eoedl"))
        XCTAssertEqual(nav.settingsPath.count, 1)
        XCTAssertEqual(nav.libraryPath.count, 0)
        nav.tab = 3
        nav.open(.aboutApp)
        nav.open(.imprint)
        XCTAssertEqual(nav.aboutPath.count, 2)
        XCTAssertEqual(nav.settingsPath.count, 1)
        XCTAssertEqual(nav.progressPath.count, 2)
        nav.back()
        XCTAssertEqual(nav.aboutPath.count, 1)
        nav.tab = 2
        nav.open(.aboutUs)
        XCTAssertEqual(nav.settingsPath.count, 2)
        XCTAssertEqual(nav.aboutPath.count, 1)
        nav.tab = 3
        nav.back(10)
        XCTAssertEqual(nav.aboutPath.count, 0)
        XCTAssertEqual(nav.settingsPath.count, 2)
    }

    func testSettingsDefaultsAndLineSpacing() {
        let d = UserDefaults(suiteName: "test.\(UUID().uuidString)")!
        let s = Settings(defaults: d)
        XCTAssertEqual(s.fontSize, Settings.standardGroesse)
        XCTAssertEqual(s.lineFactor, 1.5)
        XCTAssertEqual(s.darstellung, .system)
        // 20 pt / Faktor 1,5 → 30 pt Zeilenhöhe = natürliche 25 + 5 Zusatz
        s.fontSize = 20; s.lineFactor = 1.5
        XCTAssertEqual(s.lineSpacing, 5, accuracy: 0.01)
        s.lineFactor = 1.3
        XCTAssertEqual(s.lineSpacing, 1, accuracy: 0.01)
        XCTAssertEqual(Settings(defaults: d).fontSize, 20)
    }
